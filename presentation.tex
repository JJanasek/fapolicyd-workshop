\section{Why}

\begin{frame}{Beside SELinux, not instead}
  \begin{itemize}
    \item SELinux: \emph{behavior} (labels / domains)
    \item fapolicyd: \emph{known artifact?} (trust / provenance)
    \item Layered: SELinux is asked first; first denier wins
    \item Whitelist by trust --- not ``it has \texttt{bin\_t}''
  \end{itemize}
\end{frame}

\begin{frame}{You already know most of this}
  \centering\small
  \begin{tabular}{@{}ll@{}}
    \toprule
    \textbf{SELinux} & \textbf{fapolicyd} \\
    \midrule
    AVC record & FANOTIFY record \\
    \texttt{setenforce 0} / permissive domain & \texttt{permissive = 1} \\
    \texttt{dontaudit} & \texttt{deny} instead of \texttt{deny\_audit} \\
    \texttt{file\_contexts} + \texttt{restorecon} & trust DB + \texttt{fapolicyd-cli \dd update} \\
    policy module + \texttt{semodule} & \texttt{rules.d} drop-in + \texttt{fagenrules \dd load} \\
    allow rules: order-independent & \textbf{first match wins} --- order is policy \\
    \texttt{audit2allow} & no equivalent: add trust, do not generate rules \\
    \emph{Permission denied} (EACCES) & \emph{Operation not permitted} (EPERM) \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}{What it does \emph{not} stop}
  \begin{itemize}
    \item Shell scripts: allowed by default (\texttt{72-shell})
    \item \texttt{pattern=ld\_preload} and \texttt{pattern=static}: shipped disabled
    \item Filesystems outside \texttt{watch\_fs}, or listed in \texttt{ignore\_mounts}
    \item Trusted interpreters acting on data they are fed
    \item Root: can stop or reconfigure it (by design)
  \end{itemize}
  \vspace{0.4em}
  {\small Not a MAC substitute --- it answers ``is this file known?'', nothing more}
\end{frame}

\section{What you configure}

\begin{frame}{Four places that matter}
  \begin{enumerate}
    \item Trust DB --- known files (\texttt{rpmdb} + \texttt{trust.d/})
    \item \texttt{rules.d/} --- policy
    \item \texttt{fapolicyd.conf} --- mode / integrity / tuning
    \item \texttt{fapolicyd-filter.conf} --- what trust import includes\\
          {\small (man\,5 fapolicyd-filter.conf)}
  \end{enumerate}
\end{frame}

\begin{frame}{Hard rule}
  \centering
  \vspace{0.8em}
  {\Large Never strace / gdb the daemon}
  \vspace{0.8em}

  SIGSTOP $\Rightarrow$ every exec waits for a daemon that cannot answer\\[0.8em]
  {\small 2.0: \texttt{Type=notify} + \texttt{WatchdogSec=10s} lets systemd
  restart a stalled daemon\\[0.4em]
  \texttt{fapolicyd-selinux} hardening module:\\
  \texttt{(deny domain fapolicyd\_t (process (ptrace sigstop)))}}
\end{frame}

\begin{frame}[fragile]{Trust --- two backends, one entry format}
  \begin{itemize}
    \item \texttt{rpmdb} --- from RPM, kept current by the rpm plugin
    \item \texttt{filedb} --- what you add in \texttt{trust.d/}
  \end{itemize}
  \begin{semiverbatim}
\rootp fapolicyd-cli \dd{}dump-db | grep myapp
filedb /opt/myapp/bin/x  SIZE  HASH
  \end{semiverbatim}
  Rules only see \ok{\texttt{trust=1}} or \no{\texttt{trust=0}}\\
  {\small Hash is SHA256 or SHA512 (RPM); rule keyword \texttt{FILE\_HASH}
  replaces \texttt{SHA256HASH}}
\end{frame}

\begin{frame}{Filter (trust import scope)}
  \begin{itemize}
    \item \texttt{/etc/fapolicyd/fapolicyd-filter.conf}
    \item \texttt{+} include / \texttt{-} exclude paths (indented)
    \item Applied when trust is imported --- not per access
    \item Also: \texttt{\dd file add PATH \dd filter} (file trust)
    \item \texttt{fapolicyd-cli \dd test-filter PATH}
  \end{itemize}
\end{frame}

\begin{frame}{Rules --- recipe}
  \centering
  {\large\texttt{decision permission subject : object}}
  \vspace{0.6em}

  \textbf{First match wins}
  \vspace{0.8em}

  \raggedright
  \begin{itemize}
    \item Edit \texttt{/etc/fapolicyd/rules.d/NN-name.rules}
    \item \texttt{fagenrules} builds \texttt{compiled.rules} --- never hand-edit it
    \item Decisions: \ok{\texttt{allow}} / \no{\texttt{deny}}, with
          \texttt{\_audit}, \texttt{\_syslog} or \texttt{\_log} (both)
  \end{itemize}
\end{frame}

\end{frame}

\begin{frame}{Read the ruleset: who wins?}
  \small
  \begin{tabular}{@{}p{6.8cm}ll@{}}
    \toprule
    Attempt (as \fapop) & Result & Decided by \\
    \midrule
    Run \texttt{/usr/bin/ls} & \ok{allow} & \texttt{42-trusted-elf} \\
    Run a copy of \texttt{ls} in \texttt{/opt} & \no{deny} & \texttt{90-deny-execute} \\
    \texttt{ld.so /usr/bin/ls} & \no{deny} & \texttt{30-patterns} \\
    Run \texttt{\textasciitilde/script.sh} & \ok{allow} & \texttt{72-shell} \\
    \texttt{python3 \textasciitilde/helper.py} & \no{deny} & \texttt{70-trusted-lang} \\
    \texttt{cat \textasciitilde/notes.txt} & \ok{allow} & \texttt{95-allow-open} \\
    \midrule
    Copy of \texttt{ls} in \texttt{/opt}, as \rootp{} from bash & \ok{allow} & \texttt{20-dracut} (\texttt{uid=0 trust=1}) \\
    \bottomrule
  \end{tabular}

  \vspace{0.6em}
  That last row is why every deny lab runs as \fapop, not root
\end{frame}

\begin{frame}{Rules --- new in 2.0}
  \begin{itemize}
    \item Globs, explicit prefix: \texttt{path=glob:/opt/vendor/*/bin/tool}\\
          {\small (\texttt{exe} and \texttt{path} only; \texttt{*} does not cross \texttt{/})}
    \item Spaces: \texttt{dir="/tmp/first last/"}
    \item Packages extend languages: \texttt{\%languages+=mime/type}
    \item \texttt{dir=untrusted} deprecated --- use object \texttt{trust=}
    \item \texttt{fapolicyd-cli \dd ftype FILE} shows the MIME type rules see
  \end{itemize}
\end{frame}

\begin{frame}{Conf --- \texttt{/etc/fapolicyd/fapolicyd.conf}}
  \begin{itemize}
    \item \texttt{permissive = 1} while soaking; fix trust / rules; then \texttt{0} + restart
    \item \texttt{integrity = none | size | ima | sha256}\\
          {\small \texttt{none} is default; try \texttt{size} before \texttt{sha256} (costly)}
    \item \texttt{decision\_threads} --- parallel decisions (2.0)
    \item \texttt{ignore\_mounts} --- check with \texttt{\dd check-ignore\_mounts}
    \item \texttt{db\_max\_size = auto} --- default now
    \item \texttt{rpm\_sha256\_only}, \texttt{reset\_strategy}, \texttt{timing\_collection}
  \end{itemize}
\end{frame}

\section{Configure}

\begin{frame}[fragile]{Workflow A --- install and verify}
  \begin{semiverbatim}
\rootp dnf install fapolicyd
\rootp auditctl -a always,exit -F path=/etc/shadow -F perm=wa
\rootp systemctl enable \dd{}now fapolicyd
\rootp fapolicyd-cli \dd{}check-status
\rootp fapolicyd-cli \dd{}list | head
  \end{semiverbatim}
  \begin{itemize}
    \item \texttt{dnf} pulls the rpm plugin and \texttt{fapolicyd-selinux}\\
          {\small (F44 / 2.0.1: separate \texttt{rpm-plugin-fapolicyd};
          rawhide 2.0.2 ships it inside \texttt{fapolicyd})}
    \item Without $\ge$ 1 audit rule, \texttt{deny\_audit} is quiet --- any rule will do
    \item Expect: \texttt{Permissive: false}, trust entries $>$ 0
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Workflow B --- trust an /opt app}
  The default remediation. Prefer it over allow rules.
  \begin{semiverbatim}
\fapop /opt/myapp/bin/mytool            # \no{Operation not permitted}
\rootp fapolicyd-cli \dd{}file add /opt/myapp \dd{}trust-file myapp
\rootp fapolicyd-cli \dd{}update
\fapop /opt/myapp/bin/mytool            # \ok{runs}
  \end{semiverbatim}
  \begin{itemize}
    \item \texttt{\dd file add} writes \texttt{/etc/fapolicyd/trust.d/myapp}
    \item \texttt{\dd update} reloads the daemon's DB
    \item Inspect: \texttt{fapolicyd-cli \dd dump-db | grep myapp}
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Workflow C --- custom rule}
  Only when you need \emph{who} / \emph{how}, not just ``is it known?''
  \begin{semiverbatim}
# /etc/fapolicyd/rules.d/39-ping.rules  (one line per rule)
# BEFORE 42-trusted-elf (allow execute trust=1)
\ok{allow} perm=any gid=fapo_workshop : trust=1 path=/usr/bin/ping
\no{deny_audit} perm=execute all : trust=1 path=/usr/bin/ping
  \end{semiverbatim}
  \begin{semiverbatim}
\rootp fagenrules \dd{}load
\rootp fapolicyd-cli \dd{}check-rules \dd{}lint
\rootp fapolicyd-cli \dd{}list | grep -n ping
  \end{semiverbatim}
\end{frame}

\begin{frame}{Custom rule --- safety}
  \begin{itemize}
    \item Number \emph{before} the rule that would match first\\
          {\small (\texttt{55-} is too late: \texttt{42-trusted-elf} already allowed ping)}
    \item \texttt{systemctl reload} does \emph{not} load rules --- no \texttt{ExecReload}
    \item A bad reload keeps the previous ruleset (transactional)
    \item Test as \fapop{} --- root matches \texttt{uid=0 trust=1} first
  \end{itemize}
\end{frame}

\section{Triage}

\begin{frame}{Decision tree --- denied?}
  \begin{enumerate}
    \item Error text: \emph{Operation not permitted} (EPERM) $\rightarrow$ fapolicyd;\\
          \emph{Permission denied} (EACCES) $\rightarrow$ SELinux / DAC
    \item \texttt{ausearch -m AVC} $\rightarrow$ SELinux (label / policy)
    \item \texttt{ausearch -m FANOTIFY -i} $\rightarrow$ fapolicyd
          \begin{itemize}
            \item known software? $\rightarrow$ trust + \texttt{\dd update}
            \item who / how exception? $\rightarrow$ \texttt{rules.d} drop-in
          \end{itemize}
    \item Unclear $\rightarrow$ \texttt{deny\_log} or \texttt{\dd debug-deny}
  \end{enumerate}
\end{frame}

\begin{frame}[fragile]{Reading the records}
  Audit record (kernel $\ge$ 6.3) --- field layout:
  \begin{semiverbatim}
type=FANOTIFY resp=2 fan_type=1 fan_info=B subj_trust=2 obj_trust=0
  \end{semiverbatim}
  {\small \texttt{resp=2} deny \quad \texttt{fan\_info} rule number, \textbf{hex}
  (B = rule 11) \quad trust: 0 no, 1 yes, 2 unknown}

  \vspace{0.6em}
  Daemon line (debug mode, or syslog with \texttt{deny\_log}):
  \begin{semiverbatim}
rule=11 dec=deny_audit perm=execute auid=1001 pid=4242
  exe=/usr/bin/bash : path=/opt/x/my-ls
  ftype=application/x-executable trust=0
  \end{semiverbatim}
  {\small Same rule numbering as \texttt{fapolicyd-cli \dd list}. Values here are illustrative.}
\end{frame}

\begin{frame}{Dual stack}
  \begin{itemize}
    \item One binary can be wrong for \textbf{both} layers
    \item Wrong label $\rightarrow$ AVC first; untrusted $\rightarrow$ FANOTIFY next
    \item One attempt usually logs only the first denier
    \item Fix each layer separately (label / policy vs trust / rules)
    \item SELinux allow $\not\Rightarrow$ fapolicyd allow, and vice versa
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Debug --- two ways to get the full line}
  \textbf{Without stopping the daemon} (prefer this):
  \begin{enumerate}
    \item Change the rule's \texttt{deny\_audit} to \texttt{deny\_log}, \texttt{fagenrules \dd load}
    \item Reproduce, then \texttt{journalctl -b -u fapolicyd}
    \item Map \texttt{rule=N} with \texttt{fapolicyd-cli \dd list}; fix trust or rule order
  \end{enumerate}
  \textbf{Foreground debug} (lab only --- nothing is enforced meanwhile):
  \begin{semiverbatim}
\rootp systemctl stop fapolicyd
\rootp mkdir -p /run/fapolicyd      # 2.0.1: removed on stop
\rootp fapolicyd \dd{}permissive \dd{}debug-deny
  \end{semiverbatim}
\end{frame}

\begin{frame}{CLI cheat}
  \begin{itemize}
    \item Inspect: \texttt{\dd check-status}, \texttt{\dd list},
          \texttt{\dd dump-db}, \texttt{\dd ftype FILE}
    \item Change: \texttt{\dd file add|delete|update}, \texttt{\dd update},
          \texttt{\dd reload-rules}
    \item Check: \texttt{\dd check-rules \dd lint}, \texttt{\dd check-trustdb},
          \texttt{\dd check-path}
    \item Observe: \texttt{\dd check-metrics} (per-rule hit counts),\\
          \texttt{\dd timing-start} / \texttt{\dd timing-stop}
  \end{itemize}
\end{frame}

\begin{frame}{Anti-patterns}
  \begin{itemize}
    \item Allow-rule sprawl instead of trust
    \item Editing \texttt{compiled.rules}
    \item Testing denials as root
    \item Assuming SELinux allow $\Rightarrow$ fapolicyd allow
    \item Stopping the daemon on a real host to ``see what happens''
  \end{itemize}
\end{frame}

\section{For maintainers}

\begin{frame}{How trust stays current}
  \begin{itemize}
    \item \texttt{dnf} / \texttt{rpm} transaction $\rightarrow$ rpm plugin
          $\rightarrow$ \texttt{/run/fapolicyd/fapolicyd.fifo} $\rightarrow$ daemon
    \item Daemon rebuilds the trust DB (LMDB in \texttt{/var/lib/fapolicyd})
    \item \texttt{21-updaters.rules} exists so the updater itself is never blocked\\
          {\small (the updater waiting on a daemon that waits on the updater = deadlock)}
    \item A failed trust or rules reload keeps the active generation
    \item Plugin ownership moved: rpm's \texttt{rpm-plugin-fapolicyd} (F44)
          $\rightarrow$ \texttt{fapolicyd} itself (2.0.2)
  \end{itemize}
\end{frame}

\begin{frame}{\texttt{fapolicyd-selinux} --- our own module}
  \begin{itemize}
    \item \texttt{fapolicyd\_t}: init daemon domain, entry \texttt{fapolicyd\_exec\_t}
    \item Types: \texttt{fapolicyd\_config\_t}, \texttt{\_var\_lib\_t},
          \texttt{\_var\_run\_t}, \texttt{\_log\_t}, \texttt{\_unit\_file\_t}
    \item Broad by necessity: reads and mmaps all files, reads all domains' state,\\
          \texttt{watch\_with\_perm} on every file type
    \item Capabilities include \texttt{sys\_admin}, \texttt{sys\_ptrace}, \texttt{dac\_override}
    \item Separate \texttt{fapolicyd-hardening.cil}: nobody may ptrace or SIGSTOP the daemon
  \end{itemize}
  \vspace{0.3em}
  {\small Upstream: linux-application-whitelisting/fapolicyd-selinux}
\end{frame}

\begin{frame}{Finding your way in the source}
  \begin{itemize}
    \item \texttt{src/daemon} --- fanotify loop, reload, reports
    \item \texttt{src/library} --- rules, policy decisions, trust backends, filter
    \item \texttt{src/cli} --- \texttt{fapolicyd-cli}
    \item \texttt{src/tests} --- unit tests per component, plus \texttt{stress}
          and \texttt{perf-test}
    \item \texttt{rules.d/} --- the shipped rule files you just read
    \item \texttt{ChangeLog} --- 1.5, 1.6, 2.0 changed a lot; read it once
  \end{itemize}
\end{frame}

\section{Live}

\begin{frame}{Setup --- and why not root}
  \begin{itemize}
    \item \texttt{sudo demos/00-lab-prep.sh}
          $\rightarrow$ \texttt{fapo} in \texttt{staff\_t} (via \texttt{ssh} login)
    \item Plain \texttt{su -} stays \texttt{unconfined\_t} --- no AVC lab
    \item \texttt{id -Z} should show \texttt{staff\_t}
    \item Root + trusted shell matches \texttt{uid=0 trust=1} and is \ok{allowed}
          --- so deny as \fapop
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Deny, then fix}
  \begin{semiverbatim}
\fapop id -u                              # must NOT be 0
\rootp cp -a /usr/bin/ls /opt/fapo-workshop/my-ls
\fapop /opt/fapo-workshop/my-ls           # \no{Operation not permitted}

\rootp fapolicyd-cli \dd{}file add \\
        /opt/fapo-workshop/my-ls \dd{}trust-file workshop
\rootp fapolicyd-cli \dd{}update
\fapop /opt/fapo-workshop/my-ls           # \ok{runs}
  \end{semiverbatim}
  Diagnosed an untrusted execute, applied the standard fix, verified
\end{frame}
